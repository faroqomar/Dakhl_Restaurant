/// Opt-in test entrypoint. The native campaign controller owns preflight/build.
/// Required defines: C0_BOUNDARY_FIXTURE (fresh v4 UUID), C0_BOUNDARY_STAGE.
Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  const id = String.fromEnvironment('C0_BOUNDARY_FIXTURE');
  const stage = String.fromEnvironment('C0_BOUNDARY_STAGE');
  try {
    final support = await getApplicationSupportDirectory();
    await support.create(recursive: true);
    final fixture = await C0ImportBoundaryFixture.open(
      namespace: Directory(
        p.join(
          await support.resolveSymbolicLinks(),
          'dakhl_c0_import_boundary_qa_v1',
        ),
      ),
      fixtureId: id,
      stage: stage,
    );
    debugPrint(
      jsonEncode({
        'event': 'c0_boundary_qa_ready',
        'fixtureId': id,
        'stage': stage,
        'root': fixture.root.path,
        'pid': pid,
      }),
    );
    runApp(_BoundaryQa(fixture: fixture));
  } on Object catch (error) {
    final code = _code(error);
    debugPrint(jsonEncode({'event': 'c0_boundary_qa_refused', 'code': code}));
    runApp(
      MaterialApp(
        home: Scaffold(
          body: SafeArea(child: Text('C0 boundary QA refused: $code')),
        ),
      ),
    );
  }
}

String _code(Object error) =>
    error is C0ImportBoundaryException ? error.code : 'qa_operation_failed';

final class _BoundaryQa extends StatefulWidget {
  const _BoundaryQa({required this.fixture});
  final C0ImportBoundaryFixture fixture;
  @override
  State<_BoundaryQa> createState() => _BoundaryQaState();
}

final class _BoundaryQaState extends State<_BoundaryQa> {
  late final FileTransferCapability _transfer = widget.fixture.nativeTransfer();
  late final _PopupObserver _observer = _PopupObserver(widget.fixture);
  bool _busy = false, _cleaned = false, _verified = false, _staged = false;
  String _status = 'Ready; one native selection per fresh fixture';
  C0ImportBoundaryFixture get fixture => widget.fixture;

  Future<void> _run(Future<void> Function() action) async {
    if (_busy) return;
    setState(() => _busy = true);
    try {
      await action();
    } on Object catch (error) {
      final code = _code(error);
      debugPrint(
        jsonEncode({'event': 'c0_boundary_qa_action_failed', 'code': code}),
      );
      if (mounted) setState(() => _status = code);
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _stage() => _run(() async {
    final report = await fixture.stagePickerFiles();
    debugPrint(jsonEncode({'event': 'c0_boundary_staged', ...report}));
    if (mounted) {
      setState(() {
        _staged = true;
        _status = 'Sources staged. Capture SHA256, then use the Arabic import button.';
      });
    }
  });

  Future<void> _verify() => _run(() async {
    final report = await fixture.verify();
    debugPrint(jsonEncode({'event': 'c0_boundary_case_verified', ...report}));
    if (mounted) {
      setState(() {
        _verified = true;
        _status = 'Case verified; capture receipts and native dialog evidence.';
      });
    }
  });

  Future<void> _cleanup() => _run(() async {
    // Remove the live surface before closing its injected account owner.
    setState(() => _cleaned = true);
    await WidgetsBinding.instance.endOfFrame;
    await fixture.cleanup();
    debugPrint(
      jsonEncode({
        'event': 'c0_boundary_owned_fixture_cleaned',
        'fixtureId': fixture.fixtureId,
        'namespace': fixture.namespace.path,
      }),
    );
    if (mounted) {
      setState(
        () => _status = 'Owned root removed; compact closeout and intent tombstone retained.',
      );
    }
  });

  @override
  Widget build(BuildContext context) => MaterialApp(
    theme: DakhlTheme.light,
    locale: const Locale('ar'),
    supportedLocales: const [Locale('ar')],
    localizationsDelegates: GlobalMaterialLocalizations.delegates,
    navigatorObservers: [_observer],
    home: Scaffold(
      body: SafeArea(
        child: Column(
          children: [
            Directionality(
              textDirection: TextDirection.ltr,
              child: Padding(
                padding: const EdgeInsets.all(8),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  children: [
                    Text('C0-12 QA | ${fixture.stage} | pid $pid'),
                    Text('fixture ${fixture.fixtureId}'),
                    Text(_status),
                    SelectableText('QA_ROOT ${fixture.root.path}'),
                    const Text(
                      'Select exact-20971520.json for exact cases; '
                      'oversize-20971521.json for oversize. Dismiss the native picker '
                      'for cancel-open. Use the real Arabic confirmation dialog.',
                    ),
                    const Text(
                      'A mobile provider that cannot reach these owned files '
                      'leaves the case blocked. Provider materialization is outside '
                      'the Dart read bound.',
                    ),
                    Wrap(
                      spacing: 8,
                      children: [
                        TextButton(
                          onPressed:
                              _busy ||
                                  _cleaned ||
                                  _staged ||
                                  fixture.stage == 'cancel-open'
                              ? null
                              : _stage,
                          child: const Text('Stage owned files'),
                        ),
                        TextButton(
                          onPressed: _busy || _cleaned || _verified
                              ? null
                              : _verify,
                          child: const Text('Verify settled case'),
                        ),
                        TextButton(
                          onPressed: _busy || _cleaned ? null : _cleanup,
                          child: const Text('Cleanup owned fixture'),
                        ),
                      ],
                    ),
                  ],
                ),
              ),
            ),
            Expanded(
              child: _cleaned
                  ? const SizedBox.shrink()
                  : IgnorePointer(
                      ignoring:
                          _busy || (fixture.stage != 'cancel-open' && !_staged),
                      child: AccountsSurface(
                        model: fixture.model,
                        transfer: _transfer,
                      ),
                    ),
            ),
          ],
        ),
      ),
    ),
  );
}

final class _PopupObserver extends NavigatorObserver {
  _PopupObserver(this.fixture);
  final C0ImportBoundaryFixture fixture;
  @override
  void didPush(Route<dynamic> route, Route<dynamic>? previousRoute) {
    if (route is PopupRoute<dynamic>) fixture.recordPopup();
    super.didPush(route, previousRoute);
  }
}
