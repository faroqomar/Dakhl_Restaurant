const c0BootstrapCases = [
  'loading-absence-save-reopen',
  'corrupt-primary-explicit-recovery',
  'future-pointer-and-container',
  'orphan-and-interrupted-creation',
  'denied-root-open-visible-retry',
  'unreadable-pointer-owner-recreation',
];
const _codec = LegacyAccountCodec();

/// The same assertions run on the host and the later reviewed native target.
/// Every error comes from AppAccountsStore -> the real PowerSync owner.
Future<Map<String, Object?>> runC0BootstrapCase(
  WidgetTester tester,
  String id, {
  NativeAppReadinessFrames? nativeFrames,
}) async {
  expect(c0BootstrapCases, contains(id));
  debugPrint('C0_STEP $id fixture.create.start');
  final scope = await tester.runAsync(
    () => _Scope.create(id).timeout(const Duration(seconds: 25)),
  );
  debugPrint('C0_STEP $id fixture.create.complete');
  final s = scope!;
  addTearDown(() async {
    await tester.pumpWidget(const SizedBox.shrink());
    await tester.runAsync(s.closeAndDelete);
  });
  final result = <String, Object?>{
    'case': id,
    'status': 'passed',
    'pid': pid,
    'store': 'AppAccountsStore -> PowerSyncLocalDocumentWorkspaceStore',
    'scope': 'marked task-owned temporary root only',
  };
  ShellViewModel? shell;
  Future<void> mount(_Owner owner) async {
    debugPrint('C0_STEP $id mount.start');
    final app = DakhlRestaurantApp(
      key: UniqueKey(),
      accounts: owner.model,
      preferencesStore: s.preferences,
      onShellReady: (value) => shell = value,
    );
    if (nativeFrames == null) {
      await tester.pumpWidget(app);
    } else {
      await waitForNativeAppReadiness(
        frames: nativeFrames,
        mount: () => mountNativeAppRoot(app),
        ready: () => find.byType(DakhlShellView).evaluate().isNotEmpty,
      );
    }
    debugPrint('C0_STEP $id mount.complete');
  }

  Future<void> settled(_Owner o, AccountsState state) async {
    await _until(tester, () => o.model.state == state && !o.model.busy);
    await _until(
      tester,
      () => find.byType(AccountsSurface).evaluate().isNotEmpty,
    );
    expect(find.text(accountStatus(o.model)), findsWidgets);
    expect(tester.takeException(), isNull);
  }

  Future<void> protectedUnchanged(
    _Owner o,
    Map<String, List<int>> before,
  ) async {
    await settled(o, AccountsState.protected);
    expect(o.model.document, isNull);
    expect(o.model.canMutate, isFalse);
    expect(_button(tester, 'حفظ الحسابات').onPressed, isNull);
    await o.model.save();
    expect(o.record.mutations, 0);
    await _tap(tester, find.text('إعادة المحاولة').last);
    await settled(o, AccountsState.protected);
    expect(o.record.mutations, 0);
    expect(await tester.runAsync(() => _authorityFiles(s.accounts)), before);
  }

  if (id == c0BootstrapCases[0]) {
    final gate = Completer<void>();
    final o = s.owner(gate: gate);
    await mount(o);
    await _until(tester, () => o.record.reads == 1);
    expect(o.model.state, AccountsState.loading);
    expect(find.text('جارٍ تحميل الحسابات…'), findsWidgets);
    expect(
      tester.widget<DakhlShellView>(find.byType(DakhlShellView)).busy,
      isTrue,
    );
    expect(o.model.canMutate, isFalse);
    expect(o.model.canImport, isFalse);
    await o.model.save();
    expect(await o.model.saveShortcuts(List.filled(6, '')), isFalse);
    expect(o.record.mutations, 0);
    gate.complete();
    debugPrint('C0_STEP $id first.read.released');
    await settled(o, AccountsState.absent);
    expect(o.record.mutations, 0);
    expect(
      await tester.runAsync(
        () => File('${s.accounts.path}/active.json').exists(),
      ),
      isFalse,
    );
    await _until(tester, () => shell != null);
    shell!.setFormDraft({'quantity': '3'});
    expect(o.record.mutations, 0);
    await _tap(tester, find.text('حفظ الحسابات'));
    await settled(o, AccountsState.saved);
    await _until(
      tester,
      () => o.model.workspace.form['quantity'] == '3' && !o.model.busy,
    );
    expect(shell!.formDraft, {'quantity': '3'});
    expect(o.record.documentWrites, 1);
    final first = o.model.repository.snapshot!;
    final workspace = await tester.runAsync(
      () => o.record.readWorkspace(first.fence),
    );
    expect(workspace, isA<CapabilitySuccess<WorkspaceState>>());
    await tester.pumpWidget(const SizedBox.shrink());
    await tester.runAsync(o.close);
    final reopened = s.owner();
    await mount(reopened);
    await settled(reopened, AccountsState.saved);
    final next = reopened.model.repository.snapshot!;
    expect(_snapshot(next), _snapshot(first));
    expect(
      reopened.model.workspace.encode(),
      o.model.workspace.encode(),
      reason: 'A healthy reopen must preserve the acknowledged workspace.',
    );
    expect(reopened.record.documentWrites, 0);
    result['revision'] = next.revision;
    result['loadingMutationCalls'] = 0;
  } else if (id == c0BootstrapCases[1]) {
    final seed = await tester.runAsync(() => s.seed(twoVersions: true));
    final db = await tester.runAsync(() => _database(s.accounts));
    await tester.runAsync(
      () => db!.writeAsString('owned corrupt SQLite fixture', flush: true),
    );
    final before = (await tester.runAsync(() => _authorityFiles(s.accounts)))!;
    final o = s.owner();
    await mount(o);
    await protectedUnchanged(o, before);
    expect(o.model.failureCode, 'corrupt_primary');
    await _tap(tester, find.text('استعادة النسخة السابقة'));
    await _until(
      tester,
      () => find.text('استعادة النسخة المحلية السابقة؟').evaluate().isNotEmpty,
    );
    expect(o.model.state, AccountsState.recoveryReady);
    await _tap(
      tester,
      find.descendant(
        of: find.byType(DakhlDialog),
        matching: find.text('إلغاء'),
      ),
    );
    await settled(o, AccountsState.protected);
    expect(o.record.mutations, 0);
    expect(await tester.runAsync(() => _authorityFiles(s.accounts)), before);
    await _tap(tester, find.text('استعادة النسخة السابقة'));
    await _until(tester, () => find.byType(DakhlDialog).evaluate().isNotEmpty);
    await _tap(
      tester,
      find.descendant(
        of: find.byType(DakhlDialog),
        matching: find.text('استعادة النسخة السابقة'),
      ),
    );
    await settled(o, AccountsState.saved);
    expect(o.record.recoveries, 1);
    expect(o.model.document!.preferences.restaurant['name'], 'C0 previous');
    expect(o.model.fence, isNot(seed!.fence));
    expect(o.model.workspace.form, isEmpty);
    expect(
      await tester.runAsync(() => db!.readAsString()),
      'owned corrupt SQLite fixture',
    );
    result['cancelPreservedAuthority'] = true;
    result['freshRecoveryFence'] = true;
  } else if (id == c0BootstrapCases[2]) {
    for (final target in ['pointer', 'container']) {
      await tester.runAsync(() => s.seed());
      final file = target == 'pointer'
          ? File('${s.accounts.path}/active.json')
          : File(
              '${(await tester.runAsync(() => _database(s.accounts)))!.parent.path}/format.json',
            );
      await tester.runAsync(
        () => file.writeAsString('{"version":999}', flush: true),
      );
      final before = (await tester.runAsync(
        () => _authorityFiles(s.accounts),
      ))!;
      final o = s.owner();
      await mount(o);
      await protectedUnchanged(o, before);
      expect(o.model.failureCode, 'unsupported_format');
      await tester.pumpWidget(const SizedBox.shrink());
      await tester.runAsync(o.close);
      if (target == 'pointer') await tester.runAsync(s.resetAccounts);
    }
    result['variants'] = ['future pointer', 'future container format'];
  } else if (id == c0BootstrapCases[3]) {
    await tester.runAsync(
      () =>
          Directory('${s.accounts.path}/containers/stranded')
              .create(recursive: true),
    );
    var before = (await tester.runAsync(() => _authorityFiles(s.accounts)))!;
    final orphan = s.owner();
    await mount(orphan);
    await protectedUnchanged(orphan, before);
    expect(orphan.model.failureCode, 'incomplete_state');
    await tester.pumpWidget(const SizedBox.shrink());
    await tester.runAsync(orphan.close);
    await tester.runAsync(s.resetAccounts);
    final bytes = _codec.encode(_codec.empty());
    await tester.runAsync(() async {
      final store = _value(
        await PowerSyncLocalDocumentWorkspaceStore.open(
          directoryPath: s.accounts.path,
          checkpoint: (point) async {
            if (point == 'container.before_manifest') {
              throw StateError('owned initial prefix');
            }
          },
        ),
      );
      try {
        expect(
          await store.writeDocument(
            utf8.decode(bytes),
            writeToken: 'c0-initial',
          ),
          isA<CapabilityFailure<DocumentSnapshot>>(),
        );
      } finally {
        await store.close();
      }
    });
    before = (await tester.runAsync(() => _authorityFiles(s.accounts)))!;
    final interrupted = s.owner();
    await mount(interrupted);
    await protectedUnchanged(interrupted, before);
    expect(interrupted.model.canImport, isTrue);
    interrupted.model.prepareImport(bytes);
    await tester.pump();
    await _tap(tester, find.text('استبدال الحسابات'));
    await _until(tester, () => find.byType(DakhlDialog).evaluate().isNotEmpty);
    await _tap(
      tester,
      find.descendant(
        of: find.byType(DakhlDialog),
        matching: find.text('استبدال الحسابات'),
      ),
    );
    await settled(interrupted, AccountsState.saved);
    expect(interrupted.record.resumes, 1);
    expect(interrupted.record.documentWrites, 0);
    result['explicitValidatedImportResumed'] = true;
  } else {
    final pointerCase = id == c0BootstrapCases[5];
    final seed = await tester.runAsync(() => s.seed());
    final target = pointerCase
        ? '${s.accounts.path}/active.json'
        : s.accounts.path;
    final restoredMode = pointerCase ? 0x180 : 0x1c0; // 0600 / 0700.
    s.permissions[target] = restoredMode;
    _chmod(target, 0);
    final denied = await tester.runAsync(
      () => _denied(target, directory: !pointerCase),
    );
    if (denied != true) {
      _chmod(target, restoredMode);
      s.permissions.remove(target);
      result['status'] = 'unavailable';
      result['reason'] = 'OS did not deny the task-owned read; no simulated permission failure accepted';
      return result;
    }
    result['osDenialProvenBeforeOpen'] = true;
    final o = s.owner();
    await mount(o);
    await settled(o, AccountsState.protected);
    expect(o.model.failureCode, 'unavailable');
    expect(o.model.canMutate, isFalse);
    expect(o.record.mutations, 0);
    _chmod(target, restoredMode);
    s.permissions.remove(target);
    final before = (await tester.runAsync(() => _authorityFiles(s.accounts)))!;
    await _tap(tester, find.text('إعادة المحاولة').last);
    if (pointerCase) {
      await settled(o, AccountsState.protected);
      expect(o.record.mutations, 0);
      expect(await tester.runAsync(() => _authorityFiles(s.accounts)), before);
      await tester.pumpWidget(const SizedBox.shrink());
      await tester.runAsync(o.close);
      final recreated = s.owner();
      await mount(recreated);
      await settled(recreated, AccountsState.saved);
      expect(_snapshot(recreated.model.repository.snapshot!), _snapshot(seed!));
      expect(recreated.model.workspace.form, {'note': 'owned C0 workspace'});
      result['latchedOwnerRequiresRecreation'] = true;
    } else {
      await settled(o, AccountsState.saved);
      expect(o.opens, 2);
      expect(o.record.documentWrites, 0);
      expect(_snapshot(o.model.repository.snapshot!), _snapshot(seed!));
      expect(o.model.workspace.form, {'note': 'owned C0 workspace'});
      expect(o.record.mutations, 0);
      final restored = _value(
        (await tester.runAsync(() => o.record.readWorkspace(seed.fence)))!,
      );
      expect(restored.fence, seed.fence);
      expect(restored.snapshot!.revision, 1);
      expect(restored.snapshot!.writeToken, 'c0-workspace');
      expect(WorkspaceDocument.decode(restored.snapshot!.payload).form, {
        'note': 'owned C0 workspace',
      });
      result['visibleRetryReopenedOwner'] = true;
      result['workspaceReadOnlyPreserved'] = true;
    }
  }
  expect(tester.takeException(), isNull);
  return result;
}

DakhlButton _button(WidgetTester tester, String text) =>
    tester.widget<DakhlButton>(
      find.byWidgetPredicate((w) => w is DakhlButton && w.label == text).first,
    );
Future<void> _tap(WidgetTester tester, Finder finder) async {
  await tester.ensureVisible(finder);
  await tester.pump();
  await tester.tap(finder.hitTestable());
  await tester.pump();
}

Future<void> _until(WidgetTester tester, bool Function() ready) async {
  final clock = Stopwatch()..start();
  while (!ready()) {
    if (clock.elapsed > const Duration(seconds: 40)) {
      fail('Bounded C0 UI readiness expired');
    }
    await tester.pump(const Duration(milliseconds: 16));
    await tester.runAsync(
      () => Future<void>.delayed(const Duration(milliseconds: 10)),
    );
  }
  await tester.pump();
}

T _value<T>(CapabilityResult<T> r) {
  expect(r, isA<CapabilitySuccess<T>>());
  return (r as CapabilitySuccess<T>).value;
}

Map<String, Object?> _snapshot(DocumentSnapshot s) => {
  'payload': s.payload,
  'revision': s.revision,
  'storeId': s.fence.storeId,
  'generation': s.fence.generation,
  'writeToken': s.writeToken,
};
Future<File> _database(Directory root) async {
  final pointer =
      jsonDecode(await File('${root.path}/active.json').readAsString()) as Map;
  return File(
    '${root.path}/containers/${pointer['container']}/snapshot.sqlite',
  );
}

Future<Map<String, List<int>>> _authorityFiles(Directory root) async {
  final result = <String, List<int>>{};
  await for (final f in root.list(recursive: true, followLinks: false)) {
    if (f is! File) continue;
    final name = f.path.split('/').last;
    if ([
      'active.json',
      'initial.json',
      'previous.json',
      'format.json',
      'snapshot.sqlite',
    ].contains(name)) {
      result[f.path.substring(root.path.length)] = await f.readAsBytes();
    }
  }
  return result;
}

Future<bool> _denied(String path, {required bool directory}) async {
  try {
    if (directory) {
      await Directory(path).list().toList();
    } else {
      await File(path).readAsBytes();
    }
  } on FileSystemException catch (e) {
    return e.osError?.errorCode == 13 || e.osError?.errorCode == 1;
  }
  return false;
}

// Test-only POSIX permission control, scoped to newly created marked fixtures.
// No subprocess/OS UI driver and no dependency or production adapter changes.
void _chmod(String path, int mode) {
  final lib = Platform.isAndroid || Platform.isLinux
      ? DynamicLibrary.open(Platform.isAndroid ? 'libc.so' : 'libc.so.6')
      : DynamicLibrary.process();
  final allocate = lib
      .lookupFunction<
        Pointer<Uint8> Function(IntPtr),
        Pointer<Uint8> Function(int)
      >('malloc');
  final release = lib
      .lookupFunction<
        Void Function(Pointer<Void>),
        void Function(Pointer<Void>)
      >('free');
  final chmod = lib
      .lookupFunction<
        Int32 Function(Pointer<Uint8>, Uint32),
        int Function(Pointer<Uint8>, int)
      >('chmod');
  final bytes = utf8.encode(path);
  final p = allocate(bytes.length + 1);
  if (p == nullptr) throw StateError('chmod allocation failed');
  try {
    p.asTypedList(bytes.length + 1).setAll(0, [...bytes, 0]);
    expect(chmod(p, mode), 0, reason: 'Cannot set owned fixture permissions');
  } finally {
    release(p.cast());
  }
}

final class _Scope {
  _Scope(this.root, this.accounts, this.preferences, this.ioZone);
  final Directory root, accounts;
  final AppPreferencesStore preferences;
  final Zone ioZone;
  final owners = <_Owner>[];
  final permissions = <String, int>{};
  static Future<_Scope> create(String id) async {
    final root = await Directory.systemTemp.createTemp('dakhl-c0-protected-');
    await File('${root.path}/.c0-owned')
        .writeAsString('c0-protected-bootstrap-v1:$id', flush: true);
    final accounts = await Directory('${root.path}/accounts').create();
    final prefs = AppPreferencesStore(
      openStore: () => PowerSyncLocalPreferencesStore.open(
        databasePath: '${root.path}/shell.sqlite',
      ),
      closeStore: (s) async {
        if (s is PowerSyncLocalPreferencesStore) await s.close();
      },
    );
    final scope = _Scope(root, accounts, prefs, Zone.current);
    try {
      _value(
        await prefs.replace(
          ShellPreferences.namespace,
          jsonEncode({
            'schemaVersion': 1,
            'lastRouteId': 'settings',
            'shortcuts': [
              'sales',
              'inventory',
              'expenses',
              'cash',
              'materials',
              'settings',
            ],
          }),
        ),
      );
      return scope;
    } catch (_) {
      await scope.closeAndDelete();
      rethrow;
    }
  }

  _Owner owner({Completer<void>? gate}) {
    final value = _Owner(accounts, gate, ioZone);
    owners.add(value);
    return value;
  }

  Future<DocumentSnapshot> seed({bool twoVersions = false}) async {
    final store = _value(
      await PowerSyncLocalDocumentWorkspaceStore.open(
        directoryPath: accounts.path,
      ),
    );
    try {
      var raw = _codec.empty().toJson();
      raw = jsonDecode(jsonEncode(raw)) as Map<String, dynamic>;
      (raw['preferences'] as Map)['restaurant'] = {'name': 'C0 previous'};
      var snapshot = _value(
        await store.writeDocument(
          utf8.decode(_codec.encode(_codec.decodeJson(raw))),
          writeToken: 'c0-first',
        ),
      );
      if (twoVersions) {
        (raw['preferences'] as Map)['restaurant'] = {'name': 'C0 current'};
        snapshot = _value(
          await store.writeDocument(
            utf8.decode(_codec.encode(_codec.decodeJson(raw))),
            expected: snapshot,
            writeToken: 'c0-second',
          ),
        );
      }
      _value(
        await store.writeWorkspace(
          WorkspaceDocument(
            active: const {},
            form: const {'note': 'owned C0 workspace'},
            held: const [],
          ).encode(),
          fence: snapshot.fence,
          expectedRevision: null,
          writeToken: 'c0-workspace',
        ),
      );
      return snapshot;
    } finally {
      await store.close();
    }
  }

  Future<void> resetAccounts() async {
    await accounts.delete(recursive: true);
    await accounts.create();
  }

  Future<void> closeAndDelete() async {
    for (final e in permissions.entries) {
      _chmod(e.key, e.value);
    }
    permissions.clear();
    Object? failure;
    StackTrace? stack;
    for (final o in owners) {
      try {
        await o.close();
      } catch (e, st) {
        failure ??= e;
        stack ??= st;
      }
    }
    try {
      await preferences.close();
    } catch (e, st) {
      failure ??= e;
      stack ??= st;
    }
    if (failure != null) Error.throwWithStackTrace(failure, stack!);
    if (await root.exists()) {
      expect(await File('${root.path}/.c0-owned').exists(), isTrue);
      await root.delete(recursive: true);
    }
  }
}

final class _Owner {
  _Owner(Directory directory, Completer<void>? gate, Zone ioZone) {
    app = AppAccountsStore(
      openStore: () async {
        opens++;
        return PowerSyncLocalDocumentWorkspaceStore.open(
          directoryPath: directory.path,
        );
      },
    );
    record = _RecordedStore(app, gate, ioZone);
    model = AccountsViewModel(repository: AccountsRepository(store: record));
  }
  late final AppAccountsStore app;
  late final _RecordedStore record;
  late final AccountsViewModel model;
  int opens = 0;
  bool closed = false;
  Future<void> close() async {
    if (closed) return;
    if (record.gate != null && !record.gate!.isCompleted) {
      record.gate!.complete();
    }
    await app.close();
    model.dispose();
    closed = true;
  }
}

final class _RecordedStore implements LocalDocumentWorkspaceStore {
  _RecordedStore(this.delegate, this.gate, this.ioZone);
  final AppAccountsStore delegate;
  final Completer<void>? gate;
  // Captured inside tester.runAsync. Native SDK worker/timer initialization
  // must not inherit WidgetTester's FakeAsync clock; results remain untouched.
  final Zone ioZone;
  int reads = 0,
      documentWrites = 0,
      workspaceWrites = 0,
      recoveries = 0,
      resumes = 0,
      archives = 0;
  int get mutations =>
      documentWrites + workspaceWrites + recoveries + resumes + archives;
  @override
  Future<CapabilityResult<DocumentSnapshot?>> readDocument() =>
      ioZone.run(() async {
        reads++;
        if (reads == 1 && gate != null) await gate!.future;
        return delegate.readDocument();
      });

  @override
  Future<CapabilityResult<DocumentSnapshot>> writeDocument(
    String p, {
    required String writeToken,
    DocumentSnapshot? expected,
    bool replaceGeneration = false,
  }) {
    documentWrites++;
    return ioZone.run(
      () => delegate.writeDocument(
        p,
        writeToken: writeToken,
        expected: expected,
        replaceGeneration: replaceGeneration,
      ),
    );
  }

  @override
  Future<CapabilityResult<DocumentSnapshot>> resumeInitialCreation(
    String p, {
    required String writeToken,
  }) {
    resumes++;
    return ioZone.run(
      () => delegate.resumeInitialCreation(p, writeToken: writeToken),
    );
  }

  @override
  Future<CapabilityResult<WorkspaceState>> readWorkspace(StorageFence f) =>
      ioZone.run(() => delegate.readWorkspace(f));
  @override
  Future<CapabilityResult<WorkspaceSnapshot>> writeWorkspace(
    String p, {
    required StorageFence fence,
    required int? expectedRevision,
    required String writeToken,
  }) {
    workspaceWrites++;
    return ioZone.run(
      () => delegate.writeWorkspace(
        p,
        fence: fence,
        expectedRevision: expectedRevision,
        writeToken: writeToken,
      ),
    );
  }

  @override
  Future<CapabilityResult<PreviousDocument?>> readPrevious() =>
      ioZone.run(() => delegate.readPrevious());
  @override
  Future<CapabilityResult<String>> preserveSource(List<int> b) {
    archives++;
    return ioZone.run(() => delegate.preserveSource(b));
  }

  @override
  Future<CapabilityResult<DocumentSnapshot>> recoverPrevious(
    PreviousDocument p, {
    required String payload,
    required String writeToken,
  }) {
    recoveries++;
    return ioZone.run(
      () =>
          delegate.recoverPrevious(p, payload: payload, writeToken: writeToken),
    );
  }

  @override
  Future<CapabilityResult<void>> resolvePendingWrite() =>
      ioZone.run(() => delegate.resolvePendingWrite());
  @override
  Future<void> close() => ioZone.run(() => delegate.close());
}
