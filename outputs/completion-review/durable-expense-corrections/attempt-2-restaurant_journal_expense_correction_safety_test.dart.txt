import 'dart:convert';

import 'package:flutter_test/flutter_test.dart';
import 'package:madar_application/madar_application.dart';
import 'package:madar_domain/madar_domain.dart';
import 'package:dakhl_restaurant/accounting/journal_command_host.dart';
import 'package:dakhl_restaurant/accounting/daily_cash_queries.dart';
import 'package:dakhl_restaurant/accounting/restaurant_journal_expense_correction_commands.dart';
import 'package:dakhl_restaurant/accounting/restaurant_journal_expense_history.dart';
import 'package:dakhl_restaurant/accounting/restaurant_journal_mapper.dart';
import 'package:dakhl_restaurant/accounts/legacy_account_codec.dart';
import 'package:dakhl_restaurant/accounts/portable_journal_codec.dart';
import 'package:dakhl_restaurant/accounts/portable_journal_state.dart';

import 'restaurant_journal_expense_correction_commands_test.dart' as fixtures;

void main() {
  final mapper = RestaurantJournalMapper();
  PortableJournalState edited() {
    final state = fixtures.applyCreate(fixtures.expenseSeed());
    final command = fixtures.editExpense();
    final result = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
      command: command,
      state: state,
      mapper: mapper,
    );
    return fixtures.withExpenseReceipt(
      result.candidateState!,
      command.identity,
      result.outcomePayload!,
      2,
    );
  }

  bool valid(PortableJournalState state) =>
      RestaurantJournalExpenseHistory.validate(
        state: state,
        expenseId: 'expense',
        mapper: mapper,
      );
  for (final field in [
    'categoryId',
    'note',
    'amount',
    'createdAt',
    'funding',
    'date',
  ]) {
    test('source $field mismatch rejects before correction', () {
      final state = fixtures.applyCreate(fixtures.expenseSeed());
      final raw = jsonDecode(
        jsonEncode(state.account.toJson()),
      ) as Map<String, Object?>;
      final days = raw['days'] as Map;
      final row =
          ((days['2026-01-10'] as Map)['expenses'] as List).single as Map;
      switch (field) {
        case 'categoryId':
          row[field] = 'market';
        case 'note':
          row[field] = 'changed';
        case 'amount':
          row[field] = 101;
          row['funding'] = {
            'source': 'daily',
            'dailyAmount': 101,
            'cashAmount': 0,
          };
        case 'createdAt':
          row[field] = '2026-01-10T12:01:00.000Z';
        case 'funding':
          row[field] = {'source': 'daily', 'dailyAmount': 100, 'cashAmount': 0};
        case 'date':
          days['2026-01-11'] = days.remove('2026-01-10');
      }
      final tampered = state.copyWith(
        account: const LegacyAccountCodec().decodeJson(raw),
      );
      final result = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
        command: fixtures.editExpense(),
        state: tampered,
        mapper: mapper,
      );
      expect(result.rejected, isTrue);
      expect(result.candidateState, isNull);
    });
  }
  for (final field in [
    'amount',
    'note',
    'date',
    'profile',
    'outcome',
    'metadata',
    'policy',
    'receiptFence',
    'missingRootReceipt',
    'missingCorrectionReceipt',
    'extraReceipt',
    'fork',
  ]) {
    test('history rejects $field tampering before the next correction', () {
      final state = edited();
      final raw = jsonDecode(
        const PortableJournalCodec().encodeState(state),
      ) as Map<String, Object?>;
      // Mutate decoded portable fields through typed values to keep codec
      // validation separate from family semantic reconciliation.
      var candidate = state;
      final receipt = state.receipts.last;
      final intent = (jsonDecode(
        receipt.identity.canonicalPayloadIdentity,
      ) as Map).cast<String, Object?>();
      if (['amount', 'note', 'date', 'profile'].contains(field)) {
        if (field == 'profile') {
          (intent['profile'] as Map)['cashboxAccountId'] = 'forged:cash';
        } else {
          intent[field] = field == 'amount'
              ? '181'
              : field == 'date'
              ? '2026-01-12'
              : 'forged';
          if (field == 'amount') intent['cashAmount'] = '181';
        }
        candidate = state.copyWith(
          receipts: [
            state.receipts.first,
            DurableCommandReceipt(
              identity: DurableCommandIdentity(
                operationId: receipt.identity.operationId,
                payloadVersion: receipt.identity.payloadVersion,
                canonicalPayloadIdentity: canonicalIdentity(intent),
              ),
              status: receipt.status,
              outcomePayload: receipt.outcomePayload,
              committedRevision: receipt.committedRevision,
              sequence: receipt.sequence,
              fence: receipt.fence,
            ),
          ],
        );
      } else if (field == 'outcome') {
        final outcome =
            (jsonDecode(receipt.outcomePayload!) as Map).cast<String, Object?>()
              ..['netAmount'] = '999';
        candidate = state.copyWith(
          receipts: [
            state.receipts.first,
            DurableCommandReceipt(
              identity: receipt.identity,
              status: receipt.status,
              outcomePayload: jsonEncode(outcome),
              committedRevision: receipt.committedRevision,
              sequence: receipt.sequence,
              fence: receipt.fence,
            ),
          ],
        );
      } else if (field == 'metadata') {
        final entry = state.journals.last;
        candidate = state.copyWith(
          journals: [
            ...state.journals.take(state.journals.length - 1),
            PortableJournalEntry(
              journal: entry.journal,
              sourceType: entry.sourceType,
              sourceRecordId: entry.sourceRecordId,
              correctedFromJournalId: entry.correctedFromJournalId,
              recordedAt: entry.recordedAt,
              metadata: {...entry.metadata, 'afterSnapshot': 'forged'},
            ),
          ],
        );
      } else if (field == 'policy') {
        final entry = state.journals.last;
        final journal = BalancedJournal.validate(
          id: entry.journal.id,
          effectiveDate: entry.journal.effectiveDate,
          lines: entry.journal.lines,
          policy: RestaurantJournalPolicies.allowZero,
        );
        candidate = state.copyWith(
          journals: [
            ...state.journals.take(state.journals.length - 1),
            PortableJournalEntry(
              journal: journal,
              sourceType: entry.sourceType,
              sourceRecordId: entry.sourceRecordId,
              correctedFromJournalId: entry.correctedFromJournalId,
              recordedAt: entry.recordedAt,
              metadata: entry.metadata,
            ),
          ],
        );
      } else if (field == 'receiptFence') {
        candidate = state.copyWith(
          receipts: [
            state.receipts.first,
            DurableCommandReceipt(
              identity: receipt.identity,
              status: receipt.status,
              outcomePayload: receipt.outcomePayload,
              committedRevision: receipt.committedRevision,
              sequence: receipt.sequence,
              fence: const StorageFence(storeId: 'other', generation: 'other'),
            ),
          ],
        );
      } else if (field == 'missingRootReceipt') {
        candidate = state.copyWith(receipts: state.receipts.skip(1));
      } else if (field == 'missingCorrectionReceipt') {
        candidate = state.copyWith(receipts: state.receipts.take(1));
      } else if (field == 'extraReceipt') {
        candidate = state.copyWith(
          receipts: [
            ...state.receipts,
            DurableCommandReceipt(
              identity: fixtures
                  .editExpense(
                    operation: 'extra',
                    inverse: 'extra-inverse',
                    replacement: 'extra-replacement',
                  )
                  .identity,
              status: DurableCommandStatus.committed,
              outcomePayload: receipt.outcomePayload,
              committedRevision: 3,
              sequence: 3,
              fence: fixtures.fixtureFence,
            ),
          ],
        );
      } else if (field == 'fork') {
        final original = state.journals[1];
        final extra = RestaurantJournalExpenseHistory.correctionEntries(
          command: fixtures.editExpense(
            operation: 'fork',
            inverse: 'fork-inverse',
            replacement: 'fork-replacement',
          ),
          target: original,
          beforeSnapshot: {
            'date': '2026-01-10',
            'expense': jsonDecode(
              original.metadata['sourceSnapshot'] as String,
            ),
          },
          mapper: mapper,
        );
        candidate = state.copyWith(journals: [...state.journals, ...extra]);
      }
      expect(raw, isNotEmpty);
      expect(valid(candidate), isFalse);
      expect(
        RestaurantJournalExpenseCorrectionCoordinator.evaluate(
          command: fixtures.deleteExpense(),
          state: candidate,
          mapper: mapper,
        ).rejected,
        isTrue,
      );
    });
  }
  for (final journalId in ['original', 'replacement']) {
    test(
      'equal-total foreign accounts on $journalId cannot impersonate expense posting',
      () {
        final state = edited();
        final entries = state.journals.map((entry) {
          if (entry.journal.id != journalId) return entry;
          final changed = BalancedJournal.validate(
            id: entry.journal.id,
            effectiveDate: entry.journal.effectiveDate,
            policy: entry.journal.policy,
            lines: entry.journal.lines.map(
              (line) => JournalLine(
                accountId: 'unrelated:${line.accountId}',
                side: line.side,
                amountUnits: line.amountUnits,
                currency: line.currency,
                scale: line.scale,
              ),
            ),
          );
          return PortableJournalEntry(
            journal: changed,
            sourceType: entry.sourceType,
            sourceRecordId: entry.sourceRecordId,
            correctedFromJournalId: entry.correctedFromJournalId,
            recordedAt: entry.recordedAt,
            metadata: entry.metadata,
          );
        });
        expect(valid(state.copyWith(journals: entries)), isFalse);
      },
    );
  }
  test('custom mapper profile remaps both originals and replacements', () {
    const profile = RestaurantAccountProfile(
      dailyCashAccountId: 'custom:daily',
      cashboxAccountId: 'custom:cash',
      categoryExpensePrefix: 'custom:expense:',
    );
    final custom = RestaurantJournalMapper(profile: profile);
    final state = fixtures.applyCreate(
      fixtures.expenseSeed(mapper: custom),
      mapper: custom,
    );
    final command = fixtures.editExpense(profile: profile, category: 'market');
    final result = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
      command: command,
      state: state,
      mapper: custom,
    );
    expect(result.rejected, isFalse);
    final candidate = fixtures.withExpenseReceipt(
      result.candidateState!,
      command.identity,
      result.outcomePayload!,
      2,
    );
    expect(
      RestaurantJournalExpenseHistory.validate(
        state: candidate,
        expenseId: 'expense',
        mapper: custom,
      ),
      isTrue,
    );
    expect(valid(candidate), isFalse);
    expect(
      candidate.journals.last.journal.lines.map((l) => l.accountId),
      contains('custom:expense:market'),
    );
  });
  test('zero requires explicit allow-zero; negative cash remains source-compatible', () {
    final state = fixtures.applyCreate(fixtures.expenseSeed());
    final invalid = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
      command: fixtures.editExpense(amount: 0),
      state: state,
      mapper: mapper,
    );
    expect(invalid.rejectionCode, 'zero_value_requires_allow_zero_policy');
    final negativeCash = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
      command: fixtures.editExpense(),
      state: state,
      mapper: mapper,
    );
    expect(
      DailyCashQueries(negativeCash.candidateState!.account).cashBalance,
      BigInt.from(-180),
    );
    final zero = fixtures.editExpense(
      amount: 0,
      policy: RestaurantJournalPolicies.allowZero,
    );
    final result = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
      command: zero,
      state: state,
      mapper: mapper,
    );
    expect(result.rejected, isFalse);
    expect(
      RestaurantJournalExpenseHistory.validate(
        state: fixtures.withExpenseReceipt(
          result.candidateState!,
          zero.identity,
          result.outcomePayload!,
          2,
        ),
        expenseId: 'expense',
        mapper: mapper,
      ),
      isTrue,
    );
    final delete = fixtures.deleteExpense(
      policy: RestaurantJournalPolicies.allowZero,
    );
    final deleted = RestaurantJournalExpenseCorrectionCoordinator.evaluate(
      command: delete,
      state: fixtures.withExpenseReceipt(
        result.candidateState!,
        zero.identity,
        result.outcomePayload!,
        2,
      ),
      mapper: mapper,
    );
    expect(deleted.rejected, isFalse);
  });
  test('precutover destination, missing source, stale target and policy context fail closed', () {
    final state = fixtures.applyCreate(fixtures.expenseSeed());
    expect(
      RestaurantJournalExpenseCorrectionCoordinator.evaluate(
        command: fixtures.editExpense(target: 'missing'),
        state: state,
        mapper: mapper,
      ).rejected,
      isTrue,
    );
    expect(
      RestaurantJournalExpenseCorrectionCoordinator.evaluate(
        command: fixtures.editExpense(category: 'missing'),
        state: state,
        mapper: mapper,
      ).rejectionCode,
      'missing_category',
    );
    expect(
      RestaurantJournalExpenseCorrectionCoordinator.evaluate(
        command: fixtures.editExpense(),
        state: state,
        mapper: RestaurantJournalMapper(
          defaultPolicy: RestaurantJournalPolicies.allowZero,
        ),
      ).rejectionCode,
      'expense_policy_context_mismatch',
    );
    final command = EditOperatingExpenseCommand(
      operationId: 'before-cutover',
      accountLineage: 'lineage',
      expectedFence: fixtures.fixtureFence,
      expenseId: 'expense',
      expectedActiveJournalId: 'original',
      inverseJournalId: 'inverse',
      replacementJournalId: 'replacement',
      date: DateOnly.parse('2025-12-31', minimumYear: 1000, maximumYear: 9999),
      categoryId: 'samoon',
      amount: BigInt.from(10),
      fundingSource: FundingSource.daily,
      reason: 'change date',
      recordedAt: fixtures.fixtureStamp,
    );
    expect(
      RestaurantJournalExpenseCorrectionCoordinator.evaluate(
        command: command,
        state: state,
        mapper: mapper,
      ).rejectionCode,
      'pre_cutover_operation_rejected',
    );
    final missing = state.copyWith(journals: state.journals.take(1));
    expect(
      RestaurantJournalExpenseCorrectionCoordinator.evaluate(
        command: fixtures.editExpense(),
        state: missing,
        mapper: mapper,
      ).rejected,
      isTrue,
    );
  });
}
